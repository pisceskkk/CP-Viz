class ExampleManager:
    """示例代码块：仅用于演示 WebSlides 代码透视抽屉能力。"""

    def partition_batch(self, input_batch, batch_desc):
        # [高亮行示例] 执行核心切分逻辑
        local_batch = super().partition_batch(input_batch, batch_desc)
        padded_num_reqs = batch_desc.num_reqs
        local_batch = self._finalize_partition(input_batch, local_batch, padded_num_reqs)

        # 保存本地视图与全局采样恢复状态
        self._local_batch = local_batch
        self._sampling_global_batch = None

        # [单行高亮示例] 返回当前 rank 的局部 Batch
        return local_batch

def compute_slot_mapping_demo(token_idx: int, block_size: int, cp_size: int, interleave: int):
    """示例算子逻辑占位：演示高亮关键计算公式。"""
    virtual_block_size = block_size * cp_size

    # [高亮核心公式区间]
    virtual_block_idx = token_idx // virtual_block_size
    offset_in_vblock = token_idx % virtual_block_size
    interleave_idx = offset_in_vblock // interleave
    target_rank = interleave_idx % cp_size

    if target_rank != current_cp_rank():
        return -1  # PAD_SLOT_ID
    return resolve_physical_slot(virtual_block_idx, offset_in_vblock)
